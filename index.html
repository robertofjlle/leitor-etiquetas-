Como conectar ao Google Sheets

  1. Crie (ou abra) a planilha do Google Sheets onde quer receber os dados.
  2. No menu, vá em Extensões → Apps Script.
  3. Apague o conteúdo e cole o código abaixo.
  4. Clique em Implantar → Nova implantação, tipo App da Web.
  5. Em "Executar como": Eu. Em "Quem pode acessar": Qualquer pessoa.
  6. Implante, copie a URL gerada (termina em /exec) e cole no campo acima.
function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var data = JSON.parse(e.postData.contents);
  data.forEach(function(row) {
    sheet.appendRow([row.timestamp, row.codigo, row.descricao, row.peso, row.comprimento, row.corrida, row.lote, row.raw]);
  });
  return ContentService.createTextOutput(JSON.stringify({status:"ok"})).setMimeType(ContentService.MimeType.JSON);
}
